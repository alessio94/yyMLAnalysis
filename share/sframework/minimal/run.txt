+ImportWorkspaces {
  +MyAnalysis {
    <inputFile ="workspace.root">
  }
}

+MakeAsimovDataset {
  +MyAnalysis;
}

+Fit {
  +MyAnalysis{
    <dataset="asimovData">
  }
}

+ScanLikelihood.mu{
  +MyAnalysis {
    <datasetName = "asimovData">
    +mu {
      <min=0, max=3.5, nbins=100>
    }
  }
}

+ExportResults {
  +MyAnalysis {
    <outputFile = "result.root">
  }
}

+ExportWorkspaces {
  +MyAnalysis{
    <outputFile = "workspace-postFit.root">
  }
}
