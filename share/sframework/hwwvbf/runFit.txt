@*Calculate*/? {
  +observed {
    <datasetName="obsData">
    <invert=false>
  }
  +asimov {
    <datasetName="asimovData">
    <invert=false>
  }

  <fit.initParam.muVBF=0.8> @ ?;
  <fit.eps = 0.01> @ ?;  # (default 1.0) w/o this the fit does not converge to exactly 1 in asimov
  <fit.maxCalls = 100000> @ ?;  # (default 10 000) w/o this the fit errors are not evaluated stably
  <fit.numCPU = 1> @ ?;
  @ ? {
    <singles.SYS = "*ATLAS_*">
    <singles.THEO = "*theo_*">
    <singles.NORM = "*ATLAS_norm*">
    <singles.GGF = "muGGF">
    <singles.FAKE = "*HWW_FakeFactor_*">
    # <singles.GAMMA = "*gamma*">
  }
  <fit.startingStrategy=1,fit.printLevel=1,fit.numRetry=3, fit.numCPU = 8, fit.offset=true, fit.verbose=1> @ ?;
  <trimResult = true> @ ?;
}

@*CalculateBreakdown*/? {

# calculate the significance
+CalculateSignificance {
  +HWWVBF{
    +asimov {
      <dataset="asimovData">
    }
    +observed {
      # you need to do the observed fit in order to obtain the values of the NFs
      # you can use use blinded=true to suppress the printing of the observed mu value and significance
      <blinded=false>
      <dataset="obsData">
    }
    <fit.runHesse = true, fit.runMinos = true> @ ?;
    # running minos on all NPs is required for an accurate pull plot, but takes a while
    <fit.runMinosVars = {"*theo_*", "*ATLAS_*","*HWW_*","*mu*"}> @ ?;
    # these are technical options might help you make the fit converge if you have problems
    <fit.startingStrategy=1,fit.printLevel=1,fit.numRetry=3, fit.numCPU = 8, fit.offset=true, fit.verbose=1> @ ?;
      <fit.presetParam.muVBF=1.0> @ ?;
      <fit.initParam.muVBF=0.8> @ ?;
      # <fit.initParam.muVBF="$(muVBFInitValue)"> @ ?;
      <fit.eps = 0.01> @ ?;  # (default 1.0) w/o this the fit does not converge to exactly 1 in asimov
      <fit.maxCalls = 100000> @ ?;  # (default 10 000) w/o this the fit errors are not evaluated stably
      # (default in RooMinimizer is 500 * #parameters)
      <fit.logToFile="./workspaces/$(fitIdentifier)/fit_$(id).log"> @ ?;
  }
}
