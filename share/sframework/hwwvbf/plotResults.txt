+PlotLikelihoodScan {
 +HWWVBF {
   <arrangement="$TQPATH/../SFramework/share/templates/PlotLikelihoodScanSignificance.txt">
   <outputFile="workspaces/$(fitIdentifier)/results/likelihood_VBF.pdf">
   +asmVBF {
     <source = "LikelihoodScans/ScanLikelihood.asmVBF/Scan">
     <style.lineColor=kBlack, style.title="Asimov (#mu=1)", style.markerColor=kBlack>
   }
   +asmVBFstat {
     <source = "LikelihoodScans/ScanLikelihood.asmVBF.statonly/Scan">
     <style.lineColor=kBlack, style.title="Asimov (#mu=1, Stat.)", style.markerColor=kBlack, style.lineStyle=2>
   }
   +obsVBF {
     <source = "LikelihoodScans/ScanLikelihood.obsVBF/Scan">
     <style.lineColor=kRed+1, style.title="Observed", style.markerColor=kRed+1>
   }
   +obsVBFstat {
     <source = "LikelihoodScans/ScanLikelihood.obsVBF.statonly/Scan">
     <style.lineColor=kRed+1, style.title="Observed (Stat.)", style.markerColor=kRed+1, style.lineStyle=2>
   }
   <style.title.xAxis = "#mu_{VBF} (profiled)">
   <style.showLegend = true>
   # it's prettier without markers
   <style.markerSize=0> @ ?;
   <style.legend.x1=0.45,style.legend.x2=0.6>
 }
}
