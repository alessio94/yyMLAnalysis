# -*- mode: tqfolder -*-
#instead import a prepared one (faster than rebuilding each time!)
+ImportModels {
  +HWWVBF {
    <inputFile = "./workspaces/$(fitIdentifier)/model-raw.root", objectName="HWWVBF">
  }
}

# export our model in a simple version before any edits (needed to visualize pruning/symmetrization in plots/tables)

+ExportModels.simple {
  +HWWVBF {
    <outputFile = "./workspaces/$(fitIdentifier)/model-simple.root">
  }
}


# do some post-processing on the model
# this is some advanced magic, you're a bit on your own here.
+EditModels.postprocessing {
  +HWWVBF {
    +ImportEdits.setManualTheoryUncertainty {
      ## Uncomment following line for +/- factor 2 uncertainty in SR only
      <commands= {"@Channel.SR*/Sample.Vgamma* { +OverallSys.theo_vgamma_norm { <Val = 1., Low = 0.5, High = 2.0> }}",
        "@Channel.SR*/Sample.NonWW* { +OverallSys.theo_nonww_norm { <Val = 1., Low = 0.88, High = 1.12> }}"         
      }>
    }
  }
}


# apply pruning and symmetrization to a selected set of experimental systematics (currently: all but lumi)
+ProcessSystematics.internal {
  +HWWVBF {
    <verbose=false>
    <filters={"*"}>
    <systblacklist = {"theo_ggww*", "theo_vgamma*", "theo_nonww*"}>
    # Systematics pruning based on impact w.r.t. event yield in each region
    <SysRelThreshold = 0.001> # minimum threshold
    <SysRelCutoff = 0.8> # maximum cutoff
    <SysSigMin = 0.2, SysSigMin.select.tag="isP4Sys"> # minimum significance
    # Systematics symmetrization. Zero symmetrizes only one-sided systematics.
    <SysAsymLimit = 0.5> # only works for overall systs currently

    # symmetrize shapes that have one-sided bins
    <histoSys.systsToSymmetrizeMaxBins = {"*"}>
    <histoSys.systsToForceSymmetrizeMaxBins = {"ATLAS_JES_Flavor_Comp_top", "ATLAS_JES_PU_Rho_top"}>
    <histoSys.symmetrizeMaxBinsPrintWarnings=false>
    <histoSys.prune = true>
    <histoSys.flatChi2pValMin = 0.05>
  }
}

# export our model, both as a TQFolder and as a set of HistFactory XMLs
+ExportModels.final {
  +HWWVBF {
    <outputFile = "./workspaces/$(fitIdentifier)/model.root", writeXML="./workspaces/$(fitIdentifier)/xml">
  }
}
