# create an asimov dataset and fit it
+MakeAsimovDataset {
  +ggF_0jet;
}

+Fit.asimov {
  +ggF_0jet {
    <dataset="asimovData">
    <fit.saveSnapshot = "SnSh_AllVars_Unconditional_asimov">
    <fit.eps=0.01, fit.printLevel=1, fit.numRetry=3, fit.offset=true, fit.verbose=1>
    <fit.runHesse = true, fit.runMinos = true>
    <fit.runMinosVars = {"*"}>
    <fit.logToFile="workspaces/realistic/fit_asimov.log">
  }
}


# create breakdown and impact batch submissions

    <outputPath = "workspaces/realistic/Batch/Breakdowns">
    +asimov {
      <fit.logToFile="workspaces/realistic/fit_asimov_breakdowns_asimov.log">

      <snapshot.unconditional="SnSh_AllVars_Unconditional_asimov">
      <fit.eps=0.01, fit.printLevel=1, fit.numRetry=3, , fit.offset=true, fit.verbose=1>

  }
}
